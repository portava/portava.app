PHASE 0 — DATABASE SOURCE-OF-TRUTH RECONCILIATION
============================================================

Goal:
Prove which schema is real, which migrations are applied, whether database.types.ts matches the schema, and whether app code references tables/columns/RPC/storage buckets that do not exist.

Run:

```bash
pwd

echo "=== package scripts ==="
cat package.json | sed -n '1,260p'

echo "=== migration-related directories ==="
find . -maxdepth 5 -type d | grep -Ei "migration|supabase" | sort

echo "=== canonical migrations/ ==="
find migrations -type f 2>/dev/null | sort

echo "=== noncanonical src/migrations/ ==="
find src/migrations -type f 2>/dev/null | sort

echo "=== supabase directory ==="
find supabase -type f 2>/dev/null | sort

echo "=== database types ==="
ls -lh lib/database.types.ts 2>/dev/null || true
head -100 lib/database.types.ts 2>/dev/null || true

echo "=== all app/schema files ==="
find . -maxdepth 5 -type f $begin:math:text$ \-name \"\*\.tsx\" \-o \-name \"\*\.ts\" \-o \-name \"\*\.sql\" \-o \-name \"\*\.json\" $end:math:text$ | sort > /tmp/travel_buddy_files.txt
wc -l /tmp/travel_buddy_files.txt

echo "=== Supabase table/RPC/storage references in app code ==="
grep -RniE "from\\(['\"]|\\.from\\(['\"]|rpc\\(['\"]|storage\\.from\\(['\"]" app components lib services src 2>/dev/null | head -1000

echo "=== yesterday feature schema references ==="
grep -RniE "buddy_profiles|buddy_bookings|buddy_reviews|buddy_availability|circles|circle_members|trip_circles|event_circles|trip_crew_members|conversations|conversation_members|messages|message_translations|privacy_settings|location_shares|user_availability|activity_events|notifications|profiles|trips|events|posts" app components lib services src migrations supabase lib/database.types.ts 2>/dev/null | head -1400
